SETTINGS create table if not exists settings ( key text primary key, value text ); insert into settings (key, value) values ('shop_name', 'Festival Shop'), ('tagline', 'School Festival 2025'), ('emoji', '🎪'), ('currency', '¥'), ('tax', '0'), ('receipt_msg', 'Thank you! Come again 🎉'), ('admin_password','admin1234') on conflict (key) do nothing; -- 2. CATEGORIES create table if not exists categories ( id uuid primary key default gen_random_uuid(), name text not null, sort_order int default 0, created_at timestamptz default now() ); insert into categories (name, sort_order) values ('🍜 Noodles', 1), ('🍚 Rice', 2), ('🥗 Salads', 3), ('🥤 Drinks', 4), ('🍡 Snacks', 5), ('🍱 Sets', 6) on conflict do nothing; -- 3. MENU ITEMS (individual items + set menus) create table if not exists menu_items ( id uuid primary key default gen_random_uuid(), category_id uuid references categories(id) on delete set null, name text not null, name_jp text, -- Japanese name emoji text default '🍽', description text, price numeric not null, image_url text, -- Supabase Storage URL available boolean default true, is_set boolean default false, -- true = set menu created_at timestamptz default now() ); -- 4. SET MENU COMPONENTS (which items make up a set) create table if not exists set_components ( id uuid primary key default gen_random_uuid(), set_id uuid references menu_items(id) on delete cascade, component_id uuid references menu_items(id) on delete cascade, sort_order int default 0 ); -- 5. ORDERS (one per table session) create table if not exists orders ( id uuid primary key default gen_random_uuid(), order_number serial, table_num text not null, status text not null default 'pending' check (status in ('pending','preparing','ready','done','cancelled')), total numeric not null default 0, checked_out boolean default false, created_at timestamptz default now(), checked_out_at timestamptz ); -- 6. ORDER ITEMS create table if not exists order_items ( id uuid primary key default gen_random_uuid(), order_id uuid references orders(id) on delete cascade, menu_item_id uuid references menu_items(id) on delete set null, item_name text, item_name_jp text, item_emoji text, item_image text, is_set boolean default false, quantity int not null default 1, unit_price numeric not null, -- Kitchen tracking: comma-separated prepared flags per quantity -- e.g. "true,false,true" means item 1 and 3 are done, 2 is not prep_status text default '', -- JSON array of booleans per qty created_at timestamptz default now() ); -- 7. SET ITEM COMPONENTS snapshot (what was in the set when ordered) create table if not exists order_item_components ( id uuid primary key default gen_random_uuid(), order_item_id uuid references order_items(id) on delete cascade, component_name text, component_emoji text, component_image text );
